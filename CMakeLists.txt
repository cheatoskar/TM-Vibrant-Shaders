cmake_minimum_required(VERSION 3.21)

project(tm_vibrant_shaders VERSION 1.1.0 LANGUAGES CXX RC)

if(NOT DEFINED TM_SHADERS_VERSION)
  set(TM_SHADERS_VERSION "${PROJECT_VERSION}")
endif()
message(STATUS "TM Vibrant Shaders version: ${TM_SHADERS_VERSION}")

# TrackMania Forever is 32-bit (DirectX 9)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "TrackMania Forever is 32-bit: please configure with -A Win32.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
cmake_policy(SET CMP0091 NEW)

# Statically linked runtime (/MT) so no extra MSVC redistributable is required
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

include(FetchContent)
FetchContent_Declare(
  imgui
  GIT_REPOSITORY https://github.com/ocornut/imgui.git
  GIT_TAG v1.91.5
)
FetchContent_MakeAvailable(imgui)

configure_file(src/version.h.in "${CMAKE_CURRENT_BINARY_DIR}/tm_shaders_version.h" @ONLY)
# File version resource: also how the setup recognises its own d3d9.dll in a game folder.
string(REGEX MATCHALL "[0-9]+" TMVS_VERSION_PARTS "${TM_SHADERS_VERSION}")
list(APPEND TMVS_VERSION_PARTS 0 0 0)
list(GET TMVS_VERSION_PARTS 0 TMVS_V0)
list(GET TMVS_VERSION_PARTS 1 TMVS_V1)
list(GET TMVS_VERSION_PARTS 2 TMVS_V2)
configure_file(src/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/plugin_version.rc" @ONLY)

# Shaders are compiled at build time (fxc) and embedded as bytecode: no compile stall
# when the game starts. Keep this list in the same order as kEntryPoints in pipeline.cpp.
set(TMVS_SHADER "${CMAKE_CURRENT_SOURCE_DIR}/shaders/tmvs.hlsl")
set(TMVS_ENTRIES
  PS_LinearDepth PS_Prepare PS_DownsampleND PS_HeightMerge PS_HeightSplat PS_OcclusionShadow PS_BilateralBlur PS_ShadowHeight PS_Volumetric PS_GI
  PS_SkyClear PS_SkyStars PS_SkyBlackHole PS_SkyAurora PS_SkyRing PS_AuroraHalf PS_SkyAverage PS_Clouds PS_Reflect PS_SpillDown PS_SpillBlur PS_Lighting PS_RainDrop PS_RainSplash PS_Spray
  PS_Focus PS_DofBlur PS_Cinematic
  PS_RayMask PS_RayBlur PS_BloomDown PS_BloomUp PS_Luminance PS_Adapt PS_Final PS_FXAA PS_TAA PS_Sharpen PS_Copy
  PS_CopyDepth)

file(GLOB TMVS_SDK_BINS LIST_DIRECTORIES true "C:/Program Files (x86)/Windows Kits/10/bin/10.*")
list(SORT TMVS_SDK_BINS ORDER DESCENDING)
list(TRANSFORM TMVS_SDK_BINS APPEND "/x86")
find_program(TMVS_FXC fxc
  HINTS
    "$ENV{WindowsSdkVerBinPath}x86"
    "C:/Program Files (x86)/Windows Kits/10/bin/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/x86"
    ${TMVS_SDK_BINS}
    "C:/Program Files (x86)/Windows Kits/10/bin/x86"
  REQUIRED)

set(TMVS_SHADER_OUT "${CMAKE_CURRENT_BINARY_DIR}/shaders")
file(MAKE_DIRECTORY "${TMVS_SHADER_OUT}")
set(TMVS_SHADER_HEADERS)
set(TMVS_BYTECODE_INCLUDES "")
set(TMVS_BYTECODE_TABLE "")
foreach(entry ${TMVS_ENTRIES})
  set(out "${TMVS_SHADER_OUT}/${entry}.h")
  add_custom_command(
    OUTPUT "${out}"
    COMMAND "${TMVS_FXC}" /nologo /O3 /T ps_3_0 /E ${entry} /Vn g_${entry} /Fh "${out}" "${TMVS_SHADER}"
    DEPENDS "${TMVS_SHADER}"
    COMMENT "fxc ${entry}"
    VERBATIM)
  list(APPEND TMVS_SHADER_HEADERS "${out}")
  string(APPEND TMVS_BYTECODE_INCLUDES "#include \"shaders/${entry}.h\"
")
  string(APPEND TMVS_BYTECODE_TABLE "    {g_${entry}, sizeof(g_${entry})},
")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/tmvs_bytecode.h.in"
  "#pragma once
// Generated by CMake: precompiled ps_3_0 bytecode for every pipeline pass.
#include <windows.h>
"
  "${TMVS_BYTECODE_INCLUDES}"
  "struct TmvsBytecode { const BYTE* code; size_t size; };
"
  "static const TmvsBytecode kPrecompiledShaders[] = {
${TMVS_BYTECODE_TABLE}};
")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/tmvs_bytecode.h.in" "${CMAKE_CURRENT_BINARY_DIR}/tmvs_bytecode.h" COPYONLY)

# Rendering core shared by the plugin and the offline previewer.
add_library(tmvs_core STATIC
  src/pipeline.cpp
  src/gfx.cpp
  src/noise.cpp
  src/framecap.cpp
  src/config.cpp
  src/log.cpp
  src/weather.cpp
  ${TMVS_SHADER_HEADERS}
)
target_include_directories(tmvs_core PUBLIC src ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(tmvs_core PUBLIC d3d9 d3dcompiler dxguid)

set(TMVS_DEFINITIONS WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
set(TMVS_OPTIONS /W4 /EHsc /permissive- /utf-8)
target_compile_definitions(tmvs_core PUBLIC ${TMVS_DEFINITIONS})
target_compile_options(tmvs_core PRIVATE ${TMVS_OPTIONS})

add_library(tm_vibrant_shaders SHARED
  src/dllmain.cpp
  src/plugin.cpp
  src/engine.cpp
  src/hook.cpp
  src/depth.cpp
  src/tracer.cpp
  src/capture.cpp
  src/overlay.cpp
  src/autoquality.cpp
  src/audio.cpp
  src/proxy.cpp
  src/exports.def
  "${CMAKE_CURRENT_BINARY_DIR}/plugin_version.rc"

  ${imgui_SOURCE_DIR}/imgui.cpp
  ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp
  ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_dx9.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp
)
target_link_libraries(tm_vibrant_shaders PRIVATE tmvs_core winmm)
target_include_directories(tm_vibrant_shaders PRIVATE ${imgui_SOURCE_DIR})
target_compile_options(tm_vibrant_shaders PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc /permissive- /utf-8>)

set_target_properties(tm_vibrant_shaders PROPERTIES
  OUTPUT_NAME "TMVibrantShaders"
  PREFIX ""
)

# Offline previewer: runs the pipeline on recorded .tmcap frames (development tool).
add_executable(tmvs_preview tools/preview.cpp)
target_link_libraries(tmvs_preview PRIVATE tmvs_core)
target_compile_options(tmvs_preview PRIVATE ${TMVS_OPTIONS})

# Setup.exe: installs the files next to it (from the release zip) for the ModLoader or into
# the game folder as d3d9.dll. It deliberately carries no plugin inside itself: a setup with an
# embedded DLL looks like a dropper to virus scanners.
add_executable(tmvs_installer WIN32 tools/installer.cpp tools/installer.manifest)
target_include_directories(tmvs_installer PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(tmvs_installer PRIVATE ${TMVS_DEFINITIONS} UNICODE _UNICODE)
target_compile_options(tmvs_installer PRIVATE ${TMVS_OPTIONS})
target_link_libraries(tmvs_installer PRIVATE shell32 ole32 comctl32 version advapi32)
set_target_properties(tmvs_installer PROPERTIES OUTPUT_NAME "TM-Vibrant-Shaders-Setup")

# Release: cmake --build build --config Release --target release_zip
#   dist/TM-Vibrant-Shaders.zip
#     TM Vibrant Shaders/                 the ModLoader product folder (also the setup's source)
#       description.yaml
#       <version>/description.yaml, TMVibrantShaders.dll
#     TM-Vibrant-Shaders-Setup.exe        ModLoader or game-folder install, update, uninstall
#     README.txt, LICENSE.txt, THIRD_PARTY_NOTICES.txt
#   dist/TMVibrantShaders.dll, dist/SHA256SUMS.txt
set(TMVS_PACKAGING "${CMAKE_CURRENT_SOURCE_DIR}/packaging")
set(TMVS_DIST "${CMAKE_CURRENT_SOURCE_DIR}/dist")
set(TMVS_STAGE "${CMAKE_CURRENT_BINARY_DIR}/package")
set(TMVS_STAGE_PRODUCT "${TMVS_STAGE}/TM Vibrant Shaders")
add_custom_target(release_zip
  COMMAND ${CMAKE_COMMAND} -E rm -rf "${TMVS_STAGE}" "${TMVS_DIST}"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${TMVS_STAGE_PRODUCT}/${TM_SHADERS_VERSION}" "${TMVS_DIST}"
  COMMAND ${CMAKE_COMMAND} -E copy "${TMVS_PACKAGING}/description.yaml" "${TMVS_STAGE_PRODUCT}/description.yaml"
  COMMAND ${CMAKE_COMMAND} -E copy "${TMVS_PACKAGING}/version.yaml" "${TMVS_STAGE_PRODUCT}/${TM_SHADERS_VERSION}/description.yaml"
  COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:tm_vibrant_shaders> "${TMVS_STAGE_PRODUCT}/${TM_SHADERS_VERSION}/"
  COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:tmvs_installer> "${TMVS_STAGE}/"
  COMMAND ${CMAKE_COMMAND} -E copy "${TMVS_PACKAGING}/README.txt" "${TMVS_STAGE}/README.txt"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "${TMVS_STAGE}/LICENSE.txt"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES.md" "${TMVS_STAGE}/THIRD_PARTY_NOTICES.txt"
  COMMAND ${CMAKE_COMMAND} -E chdir "${TMVS_STAGE}"
          ${CMAKE_COMMAND} -E tar cf "${TMVS_DIST}/TM-Vibrant-Shaders.zip" --format=zip
          "TM Vibrant Shaders" "TM-Vibrant-Shaders-Setup.exe" "README.txt" "LICENSE.txt" "THIRD_PARTY_NOTICES.txt"
  COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:tm_vibrant_shaders> "${TMVS_DIST}/"
  COMMAND ${CMAKE_COMMAND} -DDIR=${TMVS_DIST} -P "${CMAKE_CURRENT_SOURCE_DIR}/packaging/sha256sums.cmake"
  DEPENDS tm_vibrant_shaders tmvs_installer
  COMMENT "Packaging dist/TM-Vibrant-Shaders.zip"
  VERBATIM)
